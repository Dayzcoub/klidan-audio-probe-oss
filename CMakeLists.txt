if(MSVC)
  set_property(TARGET KlidanAudioProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

target_link_libraries(KlidanAudioProbe PRIVATE ole32 propsys setupapi uuid avrt bcrypt)

  target_compile_options(KlidanAudioProbe PRIVATE /W4 /permissive-)

  target_compile_options(KlidanAudioProbe PRIVATE -Wall -Wextra -Wpedantic)

function(add_cli_test name case)
  add_test(NAME ${name} COMMAND ${CMAKE_COMMAND}
    "-DPROBE=$<TARGET_FILE:KlidanAudioProbe>"
    "-DCASE=${case}"
    "-DTEST_TMP_DIR=${CMAKE_CURRENT_BINARY_DIR}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/expect_cli.cmake")
endfunction()
